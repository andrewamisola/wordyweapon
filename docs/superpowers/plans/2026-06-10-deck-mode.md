> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pivot Wordy Weapon from shop/inventory management to deckbuilding with multi-strike heat combat, integrating the approved card/scoring-theater visual redesign.

**Architecture:** New logic lives in three new classic-script files (`game/deck.js`, `game/heat.js`, `game/combat-ui.js`) sharing global scope with the existing `game/script.js` monolith, which is edited only at its seams (`newEnc`, `forge`, `afterCombat`, save/load, shop routing). All work happens on the `deck-mode` git branch; `main` stays the playable classic build.

**Tech Stack:** Vanilla JS/HTML/CSS (no frameworks), plain Node scripts for unit tests and balance sim (`node tools/tests/*.js`), Electron wrapper untouched.

**Spec:** `docs/superpowers/specs/2026-06-10-deck-mode-design.md`

**Conventions for this plan:**
- The monolith is ~21k lines; line numbers drift. Every `script.js` edit step starts by **locating an anchor with Grep** (pattern given). If an anchor is missing, STOP and re-read the surrounding code rather than guessing.
- New files use top-level `const`/`function` declarations (classic scripts, shared global scope — same as `script.js`). Each ends with a `module.exports` guard so Node tests can require them.
- Unit tests are plain Node + `assert` (no framework), in `tools/tests/`. UI tasks use explicit manual verification steps instead (launch via `cd electron && npm start`, or open `game/index.html` in a browser).
- Reference visuals: `mockup-combat-redesign.html` at repo root (approved mockup). CSS is ported from it, namespaced, into `game/styles.css`.

```js
// === DECK MODE: Lexicon / Hand / Spent zones, starting decks ===
// Pure logic, no DOM. Loaded as classic script before script.js.
'use strict';

const DECK_HAND_SIZE = 8;
const DECK_DISCARDS_PER_COMBAT = 3;
const DECK_STARTING_SIZE = 14;

let _deckUidCounter = 1;

function deckCloneWord(word) {
  const c = Object.assign({}, word);
  c.uid = 'c' + (_deckUidCounter++);
  return c;
}

function deckShuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = (Math.random() * (i + 1)) | 0;
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

const DeckSys = {
  // pool: the global word array (per discovery notes); hero: hero def object.
  // 3 weapons of hero's proficiency category, 8 T1 adjectives biased to hero
  // strong elements, 2 T1 nouns, 1 flex adjective.
  buildStartingDeck(pool, hero) {
    const deck = [];
    const profCat = hero.profCat; // per notes: hero proficiency category field
    const weapons = pool.filter(w => w.type === 'weapon' && w.cat === profCat)
      .sort((a, b) => a.tier - b.tier).slice(0, 3);
    weapons.forEach(w => deck.push(deckCloneWord(w)));

    const t1adj = pool.filter(w => w.type === 'adj' && w.tier === 1);
    const strong = (hero.strong || []).slice();
    const biased = t1adj.filter(w => strong.includes(w.elem));
    const neutral = t1adj.filter(w => !strong.includes(w.elem));
    for (let i = 0; i < 6; i++) {
      const src = biased.length ? biased : t1adj;
      deck.push(deckCloneWord(src[i % src.length]));
    }
    for (let i = 0; i < 2; i++) {
      const src = neutral.length ? neutral : t1adj;
      deck.push(deckCloneWord(src[i % src.length]));
    }

    const nouns = pool.filter(w => w.type === 'noun' && w.tier === 1);
    for (let i = 0; i < 2 && nouns.length; i++) {
      deck.push(deckCloneWord(nouns[i % nouns.length]));
    }

    // flex card: one more strong-element adjective
    const flexSrc = biased.length ? biased : t1adj;
    if (flexSrc.length) deck.push(deckCloneWord(flexSrc[(deck.length) % flexSrc.length]));

    return deck.slice(0, DECK_STARTING_SIZE);
  },

  initRunDeck(S, pool, hero) {
    S.deckCards = this.buildStartingDeck(pool, hero);
  },

  addCard(S, word) {
    const c = deckCloneWord(word);
    S.deckCards.push(c);
    return c;
  },

  removeCard(S, uid) {
    S.deckCards = S.deckCards.filter(c => c.uid !== uid);
  },

  initCombat(S) {
    S.deck = {
      lexicon: deckShuffle(S.deckCards.slice()),
      hand: [],
      spent: [],
    };
    S.discardsLeft = DECK_DISCARDS_PER_COMBAT;
    this.draw(S, DECK_HAND_SIZE);
  },

  draw(S, n) {
    for (let i = 0; i < n; i++) {
      if (!S.deck.lexicon.length) {
        if (!S.deck.spent.length) break;
        S.deck.lexicon = deckShuffle(S.deck.spent.splice(0));
      }
      S.deck.hand.push(S.deck.lexicon.pop());
    }
  },

  refill(S) {
    this.draw(S, DECK_HAND_SIZE - S.deck.hand.length);
  },

  playCards(S, uids) {
    const set = new Set(uids);
    const played = S.deck.hand.filter(c => set.has(c.uid));
    S.deck.hand = S.deck.hand.filter(c => !set.has(c.uid));
    S.deck.spent.push(...played);
    return played;
  },

  // Returns false if no discards left; otherwise moves cards to spent,
  // draws the same count, decrements discardsLeft.
  discard(S, uids) {
    if (S.discardsLeft <= 0 || !uids.length) return false;
    const n = this.playCards(S, uids).length;
    this.draw(S, n);
    S.discardsLeft--;
    return true;
  },

  // Intent "scramble": random hand card back into lexicon at random position.
  scrambleOne(S) {
    if (!S.deck.hand.length) return null;
    const i = (Math.random() * S.deck.hand.length) | 0;
    const card = S.deck.hand.splice(i, 1)[0];
    S.deck.lexicon.splice((Math.random() * (S.deck.lexicon.length + 1)) | 0, 0, card);
    return card;
  },
};

if (typeof module !== 'undefined') {
  module.exports = { DeckSys, DECK_HAND_SIZE, DECK_DISCARDS_PER_COMBAT, DECK_STARTING_SIZE, deckCloneWord, deckShuffle };
}
```

// === DECK MODE: heat bands, strikes, enemy intents ===
// Pure logic, no DOM. Loaded as classic script before script.js.
'use strict';

const HEAT_BANDS = [
  { key: 'white',  name: 'WHITE HOT', mult: 2.0  },
  { key: 'yellow', name: 'YELLOW',    mult: 1.5  },
  { key: 'orange', name: 'ORANGE',    mult: 1.25 },
  { key: 'red',    name: 'RED',       mult: 1.0  },
];
const HEAT_MAX_STRIKES = 4;

const INTENT_TYPES = ['block', 'douse', 'scramble', 'strikeback', 'bolster'];
const INTENT_DEFS = {
  block:      { label: 'Inserts a pause…',  desc: 'Blocks one of your slots next strike' },
  douse:      { label: 'Douses the forge!', desc: 'Your heat will drop an extra band' },
  scramble:   { label: 'Rewrites a word…',  desc: 'A random card returns to your Lexicon' },
  strikeback: { label: 'Strikes back!',     desc: 'Deals damage to your hero' },
  bolster:    { label: 'Bolsters itself…',  desc: 'Heals a portion of its health' },
};

const HeatSys = {
  initCombat(S) {
    S.strikeNum = 1;
    S.heatShift = 0;
    S.enemyIntent = null;
  },

  bandIndex(S) {
    return Math.min((S.strikeNum - 1) + (S.heatShift || 0), HEAT_BANDS.length - 1);
  },
  band(S) { return HEAT_BANDS[this.bandIndex(S)]; },
  mult(S) { return this.band(S).mult; },
  isPerfectForge(S) { return S.strikeNum === 1; },

  // After a non-killing strike. 'cold' => combat lost (caller applies life loss).
  advance(S) {
    if (this.bandIndex(S) >= HEAT_BANDS.length - 1) return 'cold';
    S.strikeNum++;
    return 'ok';
  },

  douse(S) {
    S.heatShift = Math.min((S.heatShift || 0) + 1, HEAT_BANDS.length - 1);
  },

  // Strikeback damage scales gently with round (caller passes round).
  strikebackDamage(round) {
    return 5 + Math.floor((round || 1) / 3) * 3;
  },

  rollIntent(S, round) {
    const pool = (S.enemy && S.enemy.intents && S.enemy.intents.length)
      ? S.enemy.intents : INTENT_TYPES;
    let candidates = pool;
    if (S.enemyIntent && pool.length > 1) {
      candidates = pool.filter(t => t !== S.enemyIntent.type);
    }
    const type = candidates[(Math.random() * candidates.length) | 0];
    S.enemyIntent = {
      type,
      label: INTENT_DEFS[type].label,
      desc: INTENT_DEFS[type].desc,
      value: type === 'strikeback' ? this.strikebackDamage(round) : (type === 'bolster' ? 15 : 0),
    };
    return S.enemyIntent;
  },

  // api: { blockRandomSlot(), damageHero(n), healEnemyPct(pct), scrambleHand(), onDouse() }
  resolveIntent(S, api) {
    const it = S.enemyIntent;
    if (!it) return;
    switch (it.type) {
      case 'block':      api.blockRandomSlot(); break;
      case 'douse':      this.douse(S); api.onDouse(); break;
      case 'scramble':   api.scrambleHand(); break;
      case 'strikeback': api.damageHero(it.value); break;
      case 'bolster':    api.healEnemyPct(it.value); break;
    }
    S.enemyIntent = null;
  },
};

if (typeof module !== 'undefined') {
  module.exports = { HeatSys, HEAT_BANDS, HEAT_MAX_STRIKES, INTENT_TYPES, INTENT_DEFS };
}
```

// === Number formatting (shared, Node-testable) ===
'use strict';
function fmtBig(n) {
  n = Math.floor(n);
  const abbr = (v, suffix) => {
    const x = v >= 100 ? Math.round(v).toString() : (Math.round(v * 10) / 10).toString();
    return x.replace(/\.0$/, '') + suffix;
  };
  if (n >= 1e15) return abbr(n / 1e15, 'Q');
  if (n >= 1e12) return abbr(n / 1e12, 'T');
  if (n >= 1e9)  return abbr(n / 1e9,  'B');
  if (n >= 1e6)  return abbr(n / 1e6,  'M');
  return n.toLocaleString('en-US');
}
if (typeof module !== 'undefined') module.exports = { fmtBig };

function deckRewardWeights(round, isBossReward) {
  if (isBossReward) return { 1: 10, 2: 45, 3: 45 };
  if (round <= 6)   return { 1: 60, 2: 35, 3: 5 };
  if (round <= 15)  return { 1: 35, 2: 45, 3: 20 };
  return { 1: 15, 2: 50, 3: 35 };

function deckRollRewardCards(round, isBossReward, n = 3) {
  const weights = deckRewardWeights(round, isBossReward);
  const candidates = WORDS_POOL.filter(w => w.type !== 'gem' && w.tier >= 1); // gems stay vendor/boss items
  const out = [];
  while (out.length < n && candidates.length) {
    const tierRoll = (() => {
      const total = weights[1] + weights[2] + weights[3];
      let r = Math.random() * total;
      if ((r -= weights[1]) < 0) return 1;
      if ((r -= weights[2]) < 0) return 2;
      return 3;
    })();
    const pool = candidates.filter(w => w.tier === tierRoll && !out.includes(w));
    if (!pool.length) continue;
    out.push(pool[(Math.random() * pool.length) | 0]);
  }
  return out;
}

'use strict';
// Crude balance sim: random hands vs the HP curve across 27 rounds.
const { DeckSys } = require('../game/deck.js');
const { HeatSys, HEAT_BANDS } = require('../game/heat.js');

const BASE_HP_BLOCK0 = 50, BLOCK_GROWTH = 4.0, ROUND_MULTS = [1.0, 1.5, 2.0];
const enemyHp = r => Math.floor(BASE_HP_BLOCK0 * Math.pow(BLOCK_GROWTH, Math.floor((r - 1) / 3)) * ROUND_MULTS[(r - 1) % 3]);

// Player damage model: avg AP/W grow with deck quality per chapter; mult from
// talents grows per chapter. Tunables — adjust to taste as real data arrives.
function strikeDamage(round, strike) {
  const ch = Math.floor((round - 1) / 9);            // 0,1,2
  const ap = 6 + ch * 6 + Math.random() * 4;          // words placed * avg tier
  const w = 10 + ch * 25 + Math.random() * 10;
  const talentMult = 1 + ch * 2 + Math.random() * ch; // crude talent growth
  const heat = HEAT_BANDS[Math.min(strike - 1, 3)].mult;
  return Math.floor(ap * w * talentMult * heat);
}

const RUNS = 5000;
const strikesUsed = Array(28).fill(0), deaths = Array(28).fill(0), fights = Array(28).fill(0);
for (let run = 0; run < RUNS; run++) {
  for (let r = 1; r <= 27; r++) {
    let hp = enemyHp(r), s = 1;
    fights[r]++;
    while (s <= 4) {
      hp -= strikeDamage(r, s);
      if (hp <= 0) break;
      s++;
    }
    if (hp > 0) { deaths[r]++; } else { strikesUsed[r] += s; }
  }
}
console.log('round  avgStrikes  coldDeath%');
for (let r = 1; r <= 27; r++) {
  const kills = fights[r] - deaths[r];
  console.log(String(r).padStart(5),
    (kills ? (strikesUsed[r] / kills) : 0).toFixed(2).padStart(11),
    ((deaths[r] / fights[r]) * 100).toFixed(1).padStart(10) + '%');
}
```
